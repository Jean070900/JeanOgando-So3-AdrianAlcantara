PRÁCTICA 2 — PACEMAKER Y COROSYNC

VM1
sudo subscription-manager repos --enable=rhel-10-for-x86_64-highavailability-rpms
sudo hostnamectl set-hostname vm1
printf '192.168.0.115 vm1\n192.168.0.105 vm2\n' | sudo tee -a /etc/hosts
sudo dnf install -y pcs pacemaker resource-agents fence-agents-all
sudo firewall-cmd --permanent --add-service=high-availability
sudo firewall-cmd --reload
sudo systemctl enable --now pcsd
sudo passwd hacluster
sudo pcs host auth vm1 vm2 -u hacluster
sudo pcs cluster setup micluster --start vm1 vm2
sudo pcs cluster enable --all
sudo pcs property set stonith-enabled=false
sudo pcs resource create IP_Virtual ocf:heartbeat:IPaddr2 ip=192.168.0.200 cidr_netmask=24 op monitor interval=5s
sudo reboot

PRÁCTICA 3 — APACHE Y KEEPALIVED

VM1
sudo pcs cluster stop --all
sudo dnf install -y httpd keepalived
printf '<meta charset="UTF-8"><h1>Server1</h1>\n' | sudo tee /var/www/html/index.html
sudo systemctl enable --now httpd
sudo firewall-cmd --permanent --add-service=http
sudo firewall-cmd --permanent --add-protocol=vrrp
sudo firewall-cmd --reload
DEV=$(ip -o -4 route show default | awk '{print $5; exit}')
sudo tee /etc/keepalived/keepalived.conf >/dev/null <<EOF1
vrrp_instance VI_1 {
    state MASTER
    interface ${DEV}
    virtual_router_id 51
    priority 150
    advert_int 1
    virtual_ipaddress {
        192.168.0.201/24
    }
}
EOF1
sudo systemctl enable --now keepalived
sudo poweroff
